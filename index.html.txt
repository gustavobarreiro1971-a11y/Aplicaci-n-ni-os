function download(filename, content, mime) {
  const blob = new Blob([content], { type: mime });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = filename;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  setTimeout(() => URL.revokeObjectURL(url), 500);
}

/* Descargar PNG a partir del SVG */
function downloadPNG(filename, svgString) {
  const blob = new Blob([svgString], { type:'image/svg+xml;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  const img = new Image();
  img.onload = () => {
    // Extraer dimensiones del viewBox
    const vbMatch = svgString.match(/viewBox="0 0 (\d+) (\d+)"/);
    const w = vbMatch ? parseInt(vbMatch[1]) : 1200;
    const h = vbMatch ? parseInt(vbMatch[2]) : 400;
    const scale = w < 800 ? 2 : 1;
    const canvas = document.createElement('canvas');
    canvas.width = w * scale; canvas.height = h * scale;
    const ctx = canvas.getContext('2d');
    ctx.fillStyle = '#FFFFFF'; ctx.fillRect(0, 0, canvas.width, canvas.height);
    ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
    canvas.toBlob(b => {
      const pngUrl = URL.createObjectURL(b);
      const a = document.createElement('a');
      a.href = pngUrl; a.download = filename.replace('.svg','.png');
      document.body.appendChild(a); a.click(); document.body.removeChild(a);
      setTimeout(() => URL.revokeObjectURL(pngUrl), 500);
    });
  };
  img.src = url;
}

/* Abrir documentos en nueva pestaña (el usuario los guarda con Ctrl+S o Ctrl+P) */
function openDoc(filename) {
  const docContent = getDocContent(filename);
  const blob = new Blob([docContent], { type:'text/html' });
  const url = URL.createObjectURL(blob);
  window.open(url, '_blank');
}

/* Contenido completo de los documentos */
function getDocContent(filename) {
  if (filename === 'gb-pitch-deck.html') return PITCH_DECK_HTML;
  if (filename === 'gb-manual-marca.html') return MANUAL_HTML;
  if (filename === 'gb-redes-sociales.html') return REDES_HTML;
  return '';
}

/* Descargar todo */
function downloadAll() {
  const allFiles = [
    ...Object.entries(logos).map(([f, c]) => ({f, c, m:'image/svg+xml'})),
    {f:'gb-pitch-deck.html', c:PITCH_DECK_HTML, m:'text/html'},
    {f:'gb-manual-marca.html', c:MANUAL_HTML, m:'text/html'},
    {f:'gb-redes-sociales.html', c:REDES_HTML, m:'text/html'}
  ];
  allFiles.forEach((file, i) => {
    setTimeout(() => download(file.f, file.c, file.m), i * 250);
  });
  alert('Se descargarán ' + allFiles.length + ' archivos. Autoriza las descargas múltiples si el navegador lo pide.');
}
